cmake_minimum_required(VERSION 3.22)
project(imageheic LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
get_filename_component(IMAGE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
if(NOT ANDROID)
  set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
  set(ENABLE_CLI OFF CACHE BOOL "" FORCE)
  set(ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
  set(HIGH_BIT_DEPTH OFF CACHE BOOL "" FORCE)
  set(ENABLE_LIBNUMA OFF CACHE BOOL "" FORCE)
  add_subdirectory("${IMAGE_ROOT}/vendor/x265/source" x265-build)
  set(IMAGE_X265_BUILD "${CMAKE_CURRENT_BINARY_DIR}/x265-build")
  include(ExternalProject)
  set(IMAGE_FFMPEG "${CMAKE_CURRENT_BINARY_DIR}/ffmpeg-install")
  ExternalProject_Add(image_ffmpeg_build SOURCE_DIR "${IMAGE_ROOT}/vendor/ffmpeg-8.1.3"
    CONFIGURE_COMMAND "" BUILD_COMMAND bash "${IMAGE_ROOT}/scripts/build-video-decoder.sh"
      "${IMAGE_ROOT}/vendor/ffmpeg-8.1.3" "${CMAKE_CURRENT_BINARY_DIR}/ffmpeg-build" "${IMAGE_FFMPEG}"
    INSTALL_COMMAND "" BUILD_BYPRODUCTS "${IMAGE_FFMPEG}/lib/libavcodec.a" "${IMAGE_FFMPEG}/lib/libavutil.a")
  set(IMAGE_FFMPEG_DEP image_ffmpeg_build)
else()
  set(IMAGE_FFMPEG "${CMAKE_BINARY_DIR}/video/ffmpeg-install")
  set(IMAGE_X265_BUILD "${CMAKE_BINARY_DIR}/x265-build")
  set(IMAGE_FFMPEG_DEP softwarehevc_build)
endif()
file(MAKE_DIRECTORY "${IMAGE_FFMPEG}/include")
foreach(component avcodec avutil)
  set(FFMPEG_${component}_INCLUDE_DIR "${IMAGE_FFMPEG}/include" CACHE PATH "" FORCE)
  set(FFMPEG_${component}_LIBRARY "${IMAGE_FFMPEG}/lib/lib${component}.a" CACHE FILEPATH "" FORCE)
endforeach()
set(FFMPEG_avcodec_VERSION "62.0")
set(X265_INCLUDE_DIR "${IMAGE_ROOT}/vendor/x265/source" CACHE PATH "" FORCE)
set(X265_LIBRARY "${IMAGE_X265_BUILD}/libx265.a" CACHE FILEPATH "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
foreach(option LIBDE265 X264 OpenH264_DECODER AOM_DECODER AOM_ENCODER DAV1D SvtEnc RAV1E JPEG_DECODER JPEG_ENCODER OpenJPEG_DECODER OpenJPEG_ENCODER KVAZAAR UVG266 VVDEC VVENC OPENJPH_ENCODER LIBSHARPYUV EXAMPLES GDK_PIXBUF)
  set(WITH_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(WITH_X265 ON CACHE BOOL "" FORCE)
set(WITH_FFMPEG_DECODER ON CACHE BOOL "" FORCE)
set(WITH_FFMPEG_DECODER_PLUGIN OFF CACHE BOOL "" FORCE)
set(WITH_X265_PLUGIN OFF CACHE BOOL "" FORCE)
set(ENABLE_PLUGIN_LOADING OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_DOCUMENTATION OFF CACHE BOOL "" FORCE)
add_subdirectory("${IMAGE_ROOT}/vendor/libheif-1.23.6" heif-build)
add_dependencies(heif ${IMAGE_FFMPEG_DEP} x265-static)
target_include_directories(heif PRIVATE "${IMAGE_X265_BUILD}")
add_library(image_core STATIC image_jpeg.cpp image_metadata.cpp image_exif.cpp image_sef.cpp image_encode.cpp image_thumbnail.cpp image_verify.cpp)
add_dependencies(image_core ${IMAGE_FFMPEG_DEP})
target_include_directories(image_core PUBLIC "${IMAGE_ROOT}/vendor/libheif-1.23.6/libheif/api" "${CMAKE_CURRENT_BINARY_DIR}/heif-build" PRIVATE "${IMAGE_FFMPEG}/include")
target_link_libraries(image_core PUBLIC heif "${IMAGE_FFMPEG}/lib/libavcodec.a" "${IMAGE_FFMPEG}/lib/libavutil.a" m)
if(ANDROID)
  add_library(imageheic SHARED image_jni.cpp)
  target_link_libraries(imageheic PRIVATE image_core log)
  target_link_options(imageheic PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL" "-Wl,-Bsymbolic")
  set_target_properties(imageheic PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
else()
  add_executable(image_smoke image_smoke.cpp)
  target_link_libraries(image_smoke PRIVATE image_core pthread dl)
endif()
