cmake_minimum_required(VERSION 3.22)
project(codecprobe_hdr10 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_CLI OFF CACHE BOOL "" FORCE)
set(ENABLE_LIBNUMA OFF CACHE BOOL "" FORCE)
set(ENABLE_PIC ON CACHE BOOL "" FORCE)
# Initial diagnostic build: establish ten-bit correctness before ARM assembly tuning.
set(ENABLE_ASSEMBLY OFF CACHE BOOL "" FORCE)
set(HIGH_BIT_DEPTH ON CACHE BOOL "" FORCE)
set(MAIN12 OFF CACHE BOOL "" FORCE)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../../../../vendor/x265/source" x26510-build)
if(NOT ANDROID)
    add_executable(hdr10_smoke hdr_encoder.cpp hdr_smoke.cpp)
    target_include_directories(hdr10_smoke PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../vendor/x265/source" "${CMAKE_CURRENT_BINARY_DIR}/x26510-build")
    target_link_libraries(hdr10_smoke PRIVATE x265-static pthread dl)
endif()
