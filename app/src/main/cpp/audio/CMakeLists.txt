cmake_minimum_required(VERSION 3.22)
project(audioopus LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_BUILD_TYPE RelWithDebInfo CACHE STRING "" FORCE)
set(CMAKE_C_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_RELWITHDEBINFO "-O3 -g -DNDEBUG" CACHE STRING "" FORCE)
get_filename_component(AUDIO_VENDOR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../vendor" ABSOLUTE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_SHARED_LIBRARY OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_DRED OFF CACHE BOOL "" FORCE)
set(OPUS_OSCE OFF CACHE BOOL "" FORCE)
add_subdirectory("${AUDIO_VENDOR}/opus-1.6.1" opus-build)
foreach(flag BUILD_CXXLIBS BUILD_PROGRAMS BUILD_EXAMPLES BUILD_TESTING BUILD_DOCS WITH_OGG INSTALL_MANPAGES INSTALL_CMAKE_CONFIG_MODULE INSTALL_PKGCONFIG_MODULES ENABLE_MULTITHREADING)
  set(${flag} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${AUDIO_VENDOR}/flac-1.5.0" flac-build)
set(OPE "${AUDIO_VENDOR}/libopusenc-0.3")
add_library(opusenc STATIC ${OPE}/src/ogg_packer.c ${OPE}/src/opus_header.c ${OPE}/src/opusenc.c ${OPE}/src/picture.c ${OPE}/src/resample.c ${OPE}/src/unicode_support.c)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ope-config/config.h" "#define OPE_BUILD\n#define PACKAGE_NAME \"libopusenc\"\n#define PACKAGE_VERSION \"0.3\"\n#define RESAMPLE_FULL_SINC_TABLE 1\n#define FLOATING_POINT 1\n")
target_include_directories(opusenc PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/ope-config" "${AUDIO_VENDOR}/opus-1.6.1/include" PUBLIC "${OPE}/include")
target_compile_definitions(opusenc PRIVATE HAVE_CONFIG_H OUTSIDE_SPEEX RANDOM_PREFIX=libopusenc)
target_link_libraries(opusenc PUBLIC opus m)
add_library(audio_core STATIC audio_encoder.cpp audio_metadata.cpp)
target_link_libraries(audio_core PUBLIC opusenc FLAC)
if(ANDROID)
  add_library(audioopus SHARED audio_jni.cpp)
  target_link_libraries(audioopus PRIVATE audio_core log)
  target_link_options(audioopus PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
else()
  add_executable(audio_smoke audio_smoke.cpp)
  target_link_libraries(audio_smoke PRIVATE audio_core)
endif()
